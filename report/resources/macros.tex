% report only: internal source tags. A tag names the PAPER_AUDIT.txt entry (or
% another internal document) a number or finding comes from, and links to its
% row in the appendix "Internal sources" (\isrcrow makes the link target there).
% The tags are robust so that they survive inside \caption (a caption is also
% written to the lists of figures and tables), and they print as plain text while
% those lists are typeset, because every list entry is a link already.
\makeatletter
\newif\ifsrc@plain
\DeclareRobustCommand*{\BF}[1]{\ifsrc@plain BF\#{}#1\else\hyperlink{src:BF#1}{BF\#{}#1}\fi}
\DeclareRobustCommand*{\PW}[1]{\ifsrc@plain \#P#1\else\hyperlink{src:P#1}{\#P#1}\fi}
\DeclareRobustCommand*{\TK}[1]{\ifsrc@plain path~#1\else\hyperlink{src:TK#1}{path~#1}\fi}
\DeclareRobustCommand*{\ID}[2]{\ifsrc@plain #2\else\hyperlink{src:#1}{#2}\fi}
\AtBeginDocument{%
  \AddToHook{cmd/@starttoc/before}{\begingroup\src@plaintrue}%
  \AddToHook{cmd/@starttoc/after}{\endgroup}%
}
\makeatother
